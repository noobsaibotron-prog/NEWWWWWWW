============================================
COMANDO CMD PER BUILD RELEASE
============================================

OPZIONE 1: Usa lo script BUILD_RELEASE.bat (più semplice)
----------------------------------------------------------
cd C:\AIEQ
BUILD_RELEASE.bat

OPZIONE 2: Comando manuale completo (una riga)
----------------------------------------------------------
cd C:\AIEQ && call "C:\Program Files\Microsoft Visual Studio\2022\Community\VC\Auxiliary\Build\vcvarsall.bat" x64 && cd build && cmake -G "NMake Makefiles" -DCMAKE_BUILD_TYPE=Release -DJUCE_PATH="C:\AIEQ\JUCE" .. && cmake --build . --config Release --parallel

OPZIONE 3: Comandi separati (più leggibile)
----------------------------------------------------------
REM 1. Inizializza ambiente Visual Studio
call "C:\Program Files\Microsoft Visual Studio\2022\Community\VC\Auxiliary\Build\vcvarsall.bat" x64

REM 2. Vai nella directory build
cd C:\AIEQ\build

REM 3. Configura CMake (se non già fatto)
cmake -G "NMake Makefiles" -DCMAKE_BUILD_TYPE=Release -DJUCE_PATH="C:\AIEQ\JUCE" ..

REM 4. Compila
cmake --build . --config Release --parallel

OPZIONE 4: Se build già configurata (solo compilazione)
----------------------------------------------------------
cd C:\AIEQ\build
cmake --build . --config Release --parallel
